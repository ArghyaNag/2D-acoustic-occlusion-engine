% FRAME -- CHALLENGE 1 SOLUTION (A): DISTANCE & DIRECTION FORMULAS
%=====================================================================
\begin{frame}{Challenge 1: Distance \& Direction --- Formulas}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \flag{INVERSE-DISTANCE}
      \vspace{3pt}
      \small\textbf{Distance attenuation}
      \[
        g(d) = \frac{1.0}{1.0 + k \cdot d}
      \]
      {\scriptsize\textcolor{muted}{$k = 0.08$ -- tuned by ear on a $40\times24$ grid.}}

      \vspace{6pt}
      \flag[accent2]{STEREO PAN}
      \vspace{3pt}
      \small\textbf{Linear azimuthal panning}
      \[
        \text{pan} = \text{clip}\!\left(\frac{\Delta x}{20.0},\, -1,\, 1\right),\quad
        \Delta x = x_{\text{src}} - x_{\text{lis}}
      \]
      \[
        g_L = \tfrac{1-\text{pan}}{2}, \qquad
        g_R = \tfrac{1+\text{pan}}{2}
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \flag[accent3]{COMBINED STEREO OUTPUT}
      \vspace{3pt}
      \small\textbf{Per-sample signal vector}
      \[
        \mathbf{y}[n] = \begin{bmatrix} g_L\, g(d)\, x[n] \\ g_R\, g(d)\, x[n] \end{bmatrix}
      \]

      \vspace{6pt}
      \onslide<2->{
      \begin{notebox}[Key Properties]
        \scriptsize
        \begin{itemize}\itemsep1pt
          \item Constant total gain: $g_L + g_R = 1$ -- no volume dip at center
          \item Smooth decay: no singularity at $d{=}0$
          \item Pure NumPy: \texttt{mono * gain * pan}
        \end{itemize}
      \end{notebox}
      }
    \end{column}
  \end{columns}
\end{frame}

%=====================================================================
% FRAME -- FILTERING: ARCHITECTURE & ARTIFACT DISCOVERY
%=====================================================================
\begin{frame}{Filtering: Architecture \& Artifact Discovery}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \flag[accent3]{INITIAL ATTEMPT: PER-BLOCK FFT}
      \vspace{3pt}
      \small
      \begin{itemize}\itemsep2pt
        \item \textbf{Intuition:} multiply by smooth cutoff in frequency domain via \texttt{rfft} / \texttt{irfft}
        \item Process independent 1024-sample blocks with raised-cosine taper
        \item \textbf{Catastrophic failure:} circular convolution without overlap-add wraps energy into block edges
        \item Produces a step discontinuity every 1024 samples $\implies$ \textbf{audible 43.06\,Hz periodic buzzing artifact}!
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \flag[accent]{SOLUTION: STATEFUL BUTTERWORTH IIR}
      \vspace{3pt}
      \small
      \begin{itemize}\itemsep2pt
        \item Replaced block FFT with a causal 2nd-order Butterworth lowpass
        \item Persistent initial condition states ($\mathbf{z}_i$) carried across blocks
        \item \textbf{Bit-continuous seams:} zero step discontinuities, zero boundary buzz
      \end{itemize}
      \vspace{4pt}
      \onslide<2->{
      \begin{notebox}[Engineering Lesson]
        \scriptsize\textit{Block-based FFT without overlap-add is non-causal circular convolution. Real-time audio requires causal, state-preserving recursive filtering.}
      \end{notebox}
      }
    \end{column}
  \end{columns}
\end{frame}

%=====================================================================
% FRAME -- SECOND-ORDER BUTTERWORTH IIR LOWPASS
%=====================================================================
\begin{frame}{Second-Order Butterworth IIR Lowpass}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \flag{RECURSIVE FILTER}\;\flag[accent2]{DIFFERENCE EQUATION}
      \vspace{3pt}
      \small
      Second-order causal IIR transfer function:
      \[
        H(z) = \frac{b_0 + b_1 z^{-1} + b_2 z^{-2}}{1 + a_1 z^{-1} + a_2 z^{-2}}
      \]
      Time-domain difference equation:
      \[
        y[n] = \sum_{k=0}^{2} b_k x[n-k] - \sum_{k=1}^{2} a_k y[n-k]
      \]
      \textbf{Second-Order Sections (SOS):}
      \begin{itemize}\itemsep1pt
        \item Cascaded biquad representation for maximal numerical stability
        \item Cached filter coefficients keyed to $(f_c, f_s)$
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \flag[accent3]{ACOUSTIC CHARACTERISTICS}
      \vspace{3pt}
      \small
      \textbf{Why Butterworth?}
      \begin{itemize}\itemsep2pt
        \item \textbf{Maximally flat passband:} no ripple in audible spectrum before cutoff
        \item \textbf{Smooth roll-off:} $-12$\,dB/octave attenuation strips high-frequency treble naturally
        \item \textbf{Dynamic cutoff:} $f_c \in [300, 4000]$\,Hz dynamically adapts as geometry changes
      \end{itemize}
      \vspace{4pt}
      \onslide<2->{\flag[muted]{IIR}\;\flag[accent]{SOS FORMAT}\;\flag[accent2]{-12 dB/OCT}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Stateful Filtering: Eliminating Boundary Glitches}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}

      \vspace{2pt}
      \small

        \item \textbf{State carry-over across blocks:}
        \\[2pt]
        \centerline{\scriptsize $(y_B, \mathbf{z}_{\text{out}}) = \text{sosfilt}(\text{sos}, x_B, \text{zi}=\mathbf{z}_{\text{in}})$}

      \end{itemize}
      \vspace{2pt}
      \flag[muted]{STATEFUL ZI}\;\flag[accent3]{C0 CONT.}\;\flag[accent]{O(N) CAUSAL}

      \begin{enumerate}\itemsep1pt
        \item Echo gain: $g_{\text{echo}}\propto\alpha_r$
        \item Echo cutoff:\;
        $f_c = \max\!\big(300,\, 3500 - (1{-}\alpha_r)\!\cdot\!3000\big)$\,Hz
        \item Through-wall bleed: $\tau = (1{-}\alpha_r)$
      \end{enumerate}
      {\scriptsize\textcolor{muted}{Energy conservation: a material that transmits cannot also produce a loud reflection.}}
      \vspace{3pt}

\begin{frame}{System Architecture: Thread Isolation}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \flag{STRICT THREAD SEPARATION}
      \vspace{3pt}
      \small
      \textbf{The two real-time worlds} {\scriptsize(\texttt{shared\_state.py})}
      \begin{itemize}\itemsep2pt
        \item \textbf{Main thread} (Pygame, 60\,FPS): drag-and-drop, wall painting, grid rendering
        \item \textbf{PortAudio callback} (44.1\,kHz, $\approx$23\,ms budget): hard real-time -- must never block or allocate
      \end{itemize}
      \vspace{2pt}
      \textbf{The bridge:} \texttt{get\_snapshot()} grabs a brief lock, deep-copies positions and walls, releases immediately.
    \end{column}
    \begin{column}{0.44\textwidth}
      \flag[accent2]{DELIBERATE TRADE-OFF}
      \vspace{3pt}
      \small
      Why not share the audio thread's geometry cache?
      \begin{itemize}\itemsep2pt
        \item Cache in \texttt{filter\_state} is mutated lock-free inside the callback
        \item Sharing risks priority inversion / buffer underruns
        \item \textbf{Choice:} render thread re-evaluates the cheap geometry functions independently ($<$1\,ms)
      \end{itemize}
      \vspace{4pt}
      \onslide<2->{
      \begin{notebox}[Architecture Takeaway]
        \scriptsize\textit{Tiny duplicated CPU work in exchange for
        zero lock contention and complete thread safety.}
      \end{notebox}
      }
    \end{column}
  \end{columns}
\end{frame}
